{%- doc -%}
  Theme-built product filters (no app): a sidebar on tablet and desktop, a drawer on mobile. Groups are built
  from the products on the page: Category (product type), Colour and Finish (metaobjects), one group per
  product option (Thickness, Edge profile) and Availability. A group with fewer than two values is left out,
  because it can't narrow anything down.

  Each product in the grid carries the same values (snippets/facet-product-values.liquid). assets/facet-panel.js
  filters, counts and paginates them in the browser and keeps the state in the URL, so nothing reloads.
  Opened with the "Filter" button of snippets/facet-toolbar.liquid on mobile.

  @param {product[]} products - The products on the page (the whole collection or search results).
  @param {string} uid - A unique ID for this panel on the page, usually the section ID.
  @param {number} [show_count] - Values shown per group before "Show more". Defaults to 5.
  @param {number} [per_page] - Products per page. Defaults to 24.

  @example
  {% render 'facet-panel', products: collection.products, uid: section.id, show_count: 5, per_page: 24 %}
{%- enddoc -%}
{%- liquid
  assign show_count = show_count | default: 5
  assign per_page = per_page | default: 24
  assign drawer_id = 'FacetPanelDrawer-' | append: uid
  assign title_id = 'FacetPanelTitle-' | append: uid

  assign category_entries = ''
  assign category_hits = ''
  assign colour_entries = ''
  assign colour_hits = ''
  assign finish_entries = ''
  assign finish_hits = ''
  assign option_names = ''
  assign availability_hits = ''

  for product in products
    if product.type != blank
      assign handle = product.type | handleize
      assign category_entries = category_entries | append: handle | append: '::' | append: product.type | append: '~~'
      assign category_hits = category_hits | append: '|' | append: handle | append: '|'
    endif

    assign colour = product.metafields.custom.colour.value.name.value
    if colour != blank
      assign handle = colour | handleize
      assign colour_entries = colour_entries | append: handle | append: '::' | append: colour | append: '~~'
      assign colour_hits = colour_hits | append: '|' | append: handle | append: '|'
    endif

    assign finish = product.metafields.custom.finish.value.name.value
    if finish != blank
      assign handle = finish | handleize
      assign finish_entries = finish_entries | append: handle | append: '::' | append: finish | append: '~~'
      assign finish_hits = finish_hits | append: '|' | append: handle | append: '|'
    endif

    for option in product.options_with_values
      unless option.name == 'Title'
        assign option_names = option_names | append: option.name | append: '~~'
      endunless
    endfor

    if product.available
      assign availability_hits = availability_hits | append: '|in-stock|'
    else
      assign availability_hits = availability_hits | append: '|out-of-stock|'
    endif
  endfor

  assign category_entries = category_entries | split: '~~' | uniq | sort_natural | join: '~~'
  assign colour_entries = colour_entries | split: '~~' | uniq | sort_natural | join: '~~'
  assign finish_entries = finish_entries | split: '~~' | uniq | sort_natural | join: '~~'
  assign option_names = option_names | split: '~~' | uniq

  assign in_stock_label = 'products.facets.in_stock' | t
  assign out_of_stock_label = 'products.facets.out_of_stock' | t
  assign availability_entries = 'in-stock::' | append: in_stock_label
  if availability_hits contains '|out-of-stock|'
    assign availability_entries = availability_entries | append: '~~out-of-stock::' | append: out_of_stock_label
  endif
-%}
